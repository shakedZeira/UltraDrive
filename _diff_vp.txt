+## Front/rear drive-torque split (Vector2: front, rear) for the car's
+## drivetrain layout. RWD defaults to (0.0, 1.0) - the historical all-rear
+## flow, byte-for-byte (see the wheel loop). AWD honours
+## awd_front_torque_split (0.3 => 30% front / 70% rear); FWD is (1.0, 0.0).
+## Pure and headless-safe.
+static func drive_split_for(config: CarConfig) -> Vector2:
+    match config.drivetrain_layout:
+        CarConfig.DRIVETRAIN_FWD:
+            return Vector2(1.0, 0.0)
+        CarConfig.DRIVETRAIN_AWD:
+            var front := clampf(config.awd_front_torque_split, 0.0, 1.0)
+            return Vector2(front, 1.0 - front)
+        _:
+            return Vector2(0.0, 1.0)
+
+## Per-wheel axle torque (N*m) handed to `wheel_index` from the drivetrain's
+## TOTAL drive torque (already gear/throttle-scaled, signed by reverse and
+## engine braking): the wheel's axle gets its split fraction, halved across the
+## two wheels of that axle; an axle with zero fraction gets 0. Wheel indices
+## 0-1 are front, 2-3 rear, mirroring _wheels = [fl, fr, rl, rr]. For RWD this
+## reproduces the historical drive_torque / 2.0 on the rear wheels exactly.
+## Pure and headless-safe.
+static func wheel_drive_torque(drive_torque: float, wheel_index: int, drive_split: Vector2) -> float:
+    var fraction := drive_split.x if wheel_index < 2 else drive_split.y
+    if fraction <= 0.0:
+        return 0.0
+    return drive_torque * fraction / 2.0
+
+## Mean road speed (m/s) of the wheels the layout actually drives - the axle
+## speed the Drivetrain scales engine braking by. RWD reads only the rear
+## wheels (identical to the historical rear-axle average), FWD only the front,
+## AWD all four, so a wheelspin on whichever axle is driven snaps back. Pure
+## and headless-safe.
+static func driven_axle_speed(wheels: Array[WheelPhysics], drive_split: Vector2) -> float:
+    var spin_sum := 0.0
+    var count := 0
+    for i in range(wheels.size()):
+        var fraction := drive_split.x if i < 2 else drive_split.y
+        if fraction > 0.0:
+            spin_sum += wheels[i].wheel_angular_velocity
+            count += 1
+    if count == 0:
+        return 0.0
+    return spin_sum / float(count) * WheelPhysics.WHEEL_RADIUS
